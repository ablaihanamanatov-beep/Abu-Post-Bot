# =====================================================
# SUBSCRIBE CHECK
# =====================================================

async def check_subscription(user_id: int) -> bool:
    try:
        member = await bot.get_chat_member(
            chat_id=SUBSCRIBE_CHANNEL,
            user_id=user_id
        )

        return member.status in (
            "creator",
            "administrator",
            "member"
        )

    except TelegramBadRequest:
        return False


# =====================================================
# START
# =====================================================

@dp.message(CommandStart())
async def start(message: Message, state: FSMContext):

    await state.clear()

    add_user(message)

    if not await check_subscription(message.from_user.id):

        await message.answer(
            "👋 <b>Добро пожаловать!</b>\n\n"
            "Для использования бота необходимо "
            "подписаться на наш канал.\n\n"
            "После подписки нажмите кнопку "
            "«✅ Проверить подписку».",
            reply_markup=subscribe_keyboard()
        )
        return

    await message.answer(
        "🏠 <b>Главное меню</b>",
        reply_markup=main_menu()
    )


# =====================================================
# CHECK SUBSCRIBE
# =====================================================

@dp.callback_query(F.data == "check_subscribe")
async def callback_check_subscribe(call: CallbackQuery):

    if not await check_subscription(call.from_user.id):

        await call.answer(
            "Вы ещё не подписались.",
            show_alert=True
        )

        await call.message.edit_text(
            "❌ Вы еще не подписались на канал.",
            reply_markup=subscribe_keyboard()
        )
        return

    await call.message.edit_text(
        "✅ Подписка успешно подтверждена!"
    )

    await call.message.answer(
        "🏠 <b>Главное меню</b>",
        reply_markup=main_menu()
    )

    await call.answer()
