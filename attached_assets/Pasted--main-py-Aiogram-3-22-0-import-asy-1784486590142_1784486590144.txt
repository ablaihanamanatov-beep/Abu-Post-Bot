def add_user(message: Message):

    cursor.execute(
        "SELECT * FROM users WHERE user_id=?",
        (message.from_user.id,)
    )

    user = cursor.fetchone()

    if user:
        return

    now = datetime.now().strftime("%d.%m.%Y %H:%M")

    cursor.execute(
        """
        INSERT INTO users
        (
        user_id,
        username,
        first_name,
        register_date
        )
        VALUES
        (
        ?,?,?,?
        )
        """,
        (
            message.from_user.id,
            message.from_user.username,
            message.from_user.first_name,
            now
        )
    )

    cursor.execute(
        """
        INSERT INTO statistics
        (
        user_id
        )
        VALUES
        (
        ?
        )
        """,
        (
            message.from_user.id,
        )
    )

    db.commit()


def get_profile(user_id):

    cursor.execute(
        "SELECT * FROM users WHERE user_id=?",
        (user_id,)
    )

    return cursor.fetchone()


def increase_ads(user_id):

    cursor.execute(
        """
        UPDATE users
        SET ads_count = ads_count + 1
        WHERE user_id=?
        """,
        (
            user_id,
        )
    )

    cursor.execute(
        """
        UPDATE statistics
        SET publications = publications + 1
        WHERE user_id=?
        """,
        (
            user_id,
        )
    )

    db.commit()


def create_ad(
        user_id,
        game,
        access,
        description,
        price,
        currency,
        bank
):

    now = datetime.now().strftime("%d.%m.%Y %H:%M")

    cursor.execute(
        """
        INSERT INTO ads(
        user_id,
        game,
        access,
        description,
        price,
        currency,
        bank,
        status,
        created_at
        )
        VALUES(?,?,?,?,?,?,?,?,?)
        """,
        (
            user_id,
            game,
            access,
            description,
            price,
            currency,
            bank,
            "published",
            now
        )
    )

    db.commit()

    return cursor.lastrowid
